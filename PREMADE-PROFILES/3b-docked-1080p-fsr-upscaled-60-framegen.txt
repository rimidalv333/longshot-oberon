# Longshot Oberon: longshot_env.txt. One NAME=value per line, # for comments. Copy this file into the title folder on the
# console as longshot_env.txt (replace the old one). Every line here either matters or is meant to be changed; the rest
# are the built-in defaults.

# ---- logging ----
# 0: quiet log, worth 1-2%. Set to 1 (or remove the line) when you want a log to send with a problem.
HOS_KLOG=0

# ==== PROFILE: docked 1080p, FSR-upscaled (3D at handheld size, 1536x864), 30 fps, frame generation to 60 ====
# SAFE handheld's 3D size, but FSR 3.1 upscales it to a real 1920x1080 instead of the Lanczos filter stretching it: sharper
# on a TV, with the HUD and text drawn at native 1080p. The 3D's render targets stay at handheld size, so it should use
# only a little more memory than SAFE (the 1080p output buffers), much less than the full-1080p docked profiles.
# Frame generation adds its own memory and GPU work on top.
HOS_HANDHELD=0
HOS_OUTPUT_RES=1920x1080
HOS_PS5_DISPLAY=1920x1080
HOS_RES_SCALE=0.8

# ---- frame rate ----
# locked, evenly paced 30 (each frame shown for two refreshes). 0: unlocked
HOS_PERF_30=1

# ---- frame generation: the game at a locked 30, 60 shown (Longshot's own generator) ----
RX_VK_FRAMEGEN=1
RX_VK_MIHAWKGEN=0

# ---- picture ----
# sharpening of the 3D before the HUD, 0-100
RX_VK_UP_SHARPEN=15

# ---- performance: tested values, don't change ----
RX_VK_READBACK_LATE=newest,3980

# ---- built in (no need to set; add a line here only to change one) ----
# The tested memory and performance values are part of the title since 1.2.x: the texture budget (RX_VK_VRAM_MB=1280, max
# 1024, skip 32), the game's reserve (HOS_PS5_TEX_RESERVE_MB=768) and emergency line (HOS_PS5_MEM_LOW_MB=384), the cold store
# (HOS_PS5_SWAP_AGE=60), the import size (RX_VK_IMPORT_CHUNK_MB=2), the turn-hitch textures (RX_VK_TEX_FIRST_SMALL_MB=24,
# _CAP=256), the pipelines of areas left behind let go (RX_VK_PIPE_AREAS=3), the recorder thread and the rest. A line in
# this file wins over them; NAME=0 turns one off.

# ---- shader cache: don't remove (removing them makes the console prepare its shaders again, hours) ----
RX_VK_MESH_NOCULL=2c56dc1006942f74
RX_VK_TASK_NOOCCL=2c56dc1006942f74
