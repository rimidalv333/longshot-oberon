# ---- logging ----
# 0: quiet log, worth 1-2%. Set to 1 (or remove the line) when you want a log to send with a problem.
HOS_KLOG=0

# ==== PROFILE (proof of concept): unlocked, aiming at a native 60, no frame generation ====
# The frame rate is limited by the CPU's work per draw, not by resolution: 60 holds in light places (interiors, dungeons,
# quiet field views); busy views (Hyrule Field, Castle Town, Kakariko) run ~35-48 and will judder between 30 and 60 on a
# 60 Hz TV. The resolution is lowered only so the GPU keeps up at 60 (16.7 ms a frame): handheld, 3D at 0.75 (1152x648),
# upscaled by FSR 3.1 to 1536x864, shown at 1920x1080. For more: the mods folder's _PerfShadowLite (rename it without the
# leading _ and add _ to PerfShadow500) cuts shadow draws, and a lower field_of_view in longshot_options.ini draws less.
HOS_HANDHELD=1
HOS_RES_SCALE=0.75

# ---- frame rate ----
# unlocked (up to the TV's 60). 1: locked 30
HOS_PERF_30=0
